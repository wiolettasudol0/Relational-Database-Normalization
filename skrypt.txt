DROP TABLE IF EXISTS REZERWACJE_POKOJE, REZERWACJE, POKOJE, TYPY_POKOJU, KLIENCI;

CREATE TABLE TYPY_POKOJU (
    TYP_POKOJU VARCHAR(50) PRIMARY KEY,
    LOZKA_1OS INT NOT NULL,
    LOZKA_2OS INT NOT NULL,
    KLIMATYZACJA BOOLEAN NOT NULL
);

CREATE TABLE POKOJE (
    NUMER_POKOJU INT PRIMARY KEY,
    PIETRO INT NOT NULL,
    TYP_POKOJU VARCHAR(50) NOT NULL,
    FOREIGN KEY (TYP_POKOJU) REFERENCES TYPY_POKOJU(TYP_POKOJU)
);

CREATE TABLE KLIENCI (
    ID_KLIENTA INT AUTO_INCREMENT PRIMARY KEY,
    NAZWISKO VARCHAR(50) NOT NULL
);

CREATE TABLE REZERWACJE (
    NUMER_REZERWACJI CHAR(6) PRIMARY KEY,
    ID_KLIENTA INT NOT NULL,
    LICZBA_OSOB INT NOT NULL,
    LICZBA_POKOI INT NOT NULL,
    FOREIGN KEY (ID_KLIENTA) REFERENCES KLIENCI(ID_KLIENTA)
);

CREATE TABLE REZERWACJE_POKOJE (
    NUMER_REZERWACJI CHAR(6) NOT NULL,
    NUMER_POKOJU INT NOT NULL,
    DATA_ZAKWATEROWANIA DATE NOT NULL,
    DATA_WYKWATEROWANIA DATE NOT NULL,
    PRIMARY KEY (NUMER_REZERWACJI, NUMER_POKOJU),
    FOREIGN KEY (NUMER_REZERWACJI) REFERENCES REZERWACJE(NUMER_REZERWACJI),
    FOREIGN KEY (NUMER_POKOJU) REFERENCES POKOJE(NUMER_POKOJU)
);
